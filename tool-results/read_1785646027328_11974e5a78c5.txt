   260→// === Client Dashboard ===
   261→function ClientDashboard({ onNewRequest, onViewRequest, onNavigate }: { onNewRequest: () => void; onViewRequest: (id: string) => void; onNavigate: (v: View) => void }) {
   262→  const { user } = useAuth()
   263→  const { data: requests, isLoading } = useQuery<ImportRequest[]>({
   264→    queryKey: ['client-requests'],
   265→    queryFn: async () => {
   266→      const res = await fetch('/api/requests')
   267→      if (!res.ok) return []
   268→      const data = await res.json()
   269→      return Array.isArray(data) ? data : []
   270→    },
   271→  })
   272→
   273→  const reqList = requests ?? []
   274→  const stats = {
   275→    total: reqList.length,
   276→    active: reqList.filter((r) => !['ENTREGADO', 'CERRADO'].includes(r.status)).length,
   277→    delivered: reqList.filter((r) => r.status === 'ENTREGADO').length,
   278→    new: reqList.filter((r) => r.status === 'NUEVA').length,
   279→  }
   280→
   281→  return (
   282→    <div className="space-y-6">
   283→      <div className="rounded-2xl bg-gradient-to-br from-primary/10 via-emerald-500/5 to-transparent p-6">
   284→        <h2 className="text-2xl font-bold">¡Hola, {user?.firstName}! 👋</h2>
   285→        <p className="mt-1 text-muted-foreground">Importa cualquier producto desde China. Explora nuestro catálogo o solicita algo personalizado.</p>
   286→        <div className="mt-4 flex gap-2">
   287→          <Button onClick={() => onNavigate('catalog')} className="gap-1.5"><ShoppingBag className="h-4 w-4" /> Explorar catálogo</Button>
   288→          <Button variant="outline" onClick={onNewRequest} className="gap-1.5"><Plus className="h-4 w-4" /> Solicitar producto personalizado</Button>
   289→        </div>
   290→      </div>
   291→
   292→      <motion.div variants={staggerContainer} initial="hidden" animate="visible" className="grid grid-cols-2 gap-4 lg:grid-cols-4">
   293→        <motion.div variants={staggerItem}><Card><CardContent className="p-4"><p className="text-xs text-muted-foreground">Total solicitudes</p><p className="mt-1 text-2xl font-bold"><AnimatedCounter value={stats.total} /></p></CardContent></Card></motion.div>
   294→        <motion.div variants={staggerItem}><Card><CardContent className="p-4"><p className="text-xs text-muted-foreground">En proceso</p><p className="mt-1 text-2xl font-bold text-amber-600"><AnimatedCounter value={stats.active} /></p></CardContent></Card></motion.div>
   295→        <motion.div variants={staggerItem}><Card><CardContent className="p-4"><p className="text-xs text-muted-foreground">Entregados</p><p className="mt-1 text-2xl font-bold text-emerald-600"><AnimatedCounter value={stats.delivered} /></p></CardContent></Card></motion.div>
   296→        <motion.div variants={staggerItem}><Card><CardContent className="p-4"><p className="text-xs text-muted-foreground">Nuevos</p><p className="mt-1 text-2xl font-bold text-sky-600"><AnimatedCounter value={stats.new} /></p></CardContent></Card></motion.div>
   297→      </motion.div>
   298→
   299→      <div>
   300→        <div className="mb-3 flex items-center justify-between">
   301→          <h3 className="text-lg font-semibold">Solicitudes recientes</h3>
   302→          <Button variant="ghost" size="sm" onClick={onNewRequest} className="gap-1 text-xs"><Plus className="h-3 w-3" /> Nueva</Button>
   303→        </div>
   304→        {isLoading ? (
   305→          <div className="space-y-2">{Array.from({ length: 3 }).map((_, i) => <Skeleton key={i} className="h-20 w-full" />)}</div>
   306→        ) : reqList.length === 0 ? (
   307→          <Card><CardContent className="flex flex-col items-center justify-center py-12 text-center">
   308→            <Package className="h-10 w-10 text-muted-foreground/40" />
   309→            <p className="mt-3 text-sm font-medium">Aún no tienes solicitudes</p>
   310→            <p className="text-xs text-muted-foreground">Explora el catálogo o solicita un producto personalizado</p>
   311→            <div className="mt-4 flex gap-2">
   312→              <Button variant="outline" size="sm" onClick={() => onNavigate('catalog')} className="gap-1.5"><ShoppingBag className="h-4 w-4" /> Ver catálogo</Button>
   313→              <Button size="sm" onClick={onNewRequest} className="gap-1.5"><Plus className="h-4 w-4" /> Nueva solicitud</Button>
   314→            </div>
   315→          </CardContent></Card>
   316→        ) : (
   317→          <div className="space-y-2">
   318→            {reqList.slice(0, 5).map((r) => (
   319→              <button key={r.id} onClick={() => onViewRequest(r.id)} className="flex w-full items-center gap-3 rounded-xl border bg-card p-4 text-left transition-colors hover:bg-muted/40">
   320→                <div className="min-w-0 flex-1">
   321→                  <div className="flex items-center gap-2">
   322→                    <code className="text-xs font-medium text-primary">{r.number}</code>
   323→                    <Badge variant="outline" className={cn('text-[10px]', REQUEST_STATUS_COLORS[r.status])}>{REQUEST_STATUS_LABELS[r.status]}</Badge>
   324→                  </div>
   325→                  <p className="mt-1 truncate text-sm font-medium">{r.productName}</p>
   326→                  <p className="text-xs text-muted-foreground">{r.quantity} unidades · {timeAgo(r.createdAt)}</p>
   327→                </div>
   328→                <ChevronRight className="h-4 w-4 text-muted-foreground" />
   329→              </button>
   330→            ))}
   331→          </div>
   332→        )}
   333→      </div>
   334→    </div>
   335→  )

   339→function ClientCatalog({ onProductClick }: { onProductClick: (p: Product) => void }) {
   340→  const [query, setQuery] = useState('')
   341→  const [category, setCategory] = useState('all')
   342→  const { data: products, isLoading } = useQuery<Product[]>({
   343→    queryKey: ['products-catalog'],
   344→    queryFn: async () => (await fetch('/api/products')).json(),
   345→  })
   346→
   347→  const categories = products ? [...new Set(products.map((p) => p.category?.name).filter(Boolean))] : []
   348→  const filtered = (products ?? []).filter((p) => {
   349→    const matchesQuery = !query || p.name.toLowerCase().includes(query.toLowerCase()) || p.description?.toLowerCase().includes(query.toLowerCase())
   350→    const matchesCat = category === 'all' || p.category?.name === category
   351→    return matchesQuery && matchesCat
   352→  })
   353→
   354→  return (
   355→    <div className="space-y-6">
   356→      {/* Hero banner */}
   357→      <div className="rounded-2xl bg-gradient-to-br from-primary/10 via-emerald-500/5 to-transparent p-6">
   358→        <h2 className="text-xl font-bold">Catálogo de productos importables 🛍️</h2>
   359→        <p className="mt-1 text-sm text-muted-foreground">Explora los productos que podemos importar para ti desde China. Click en cualquier producto para solicitar una importación.</p>
   360→      </div>
   361→
   362→      {/* Search + filters */}
   363→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center">
   364→        <div className="relative max-w-sm flex-1">
   365→          <Search className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   366→          <Input placeholder="Buscar productos..." value={query} onChange={(e) => setQuery(e.target.value)} className="pl-9" />
   367→        </div>
   368→        <div className="flex flex-wrap gap-1.5">
   369→          <Button variant={category === 'all' ? 'default' : 'outline'} size="sm" onClick={() => setCategory('all')}>Todos</Button>
   370→          {categories.map((c) => (
   371→            <Button key={c} variant={category === c ? 'default' : 'outline'} size="sm" onClick={() => setCategory(c as string)}>{c}</Button>
   372→          ))}
   373→        </div>
   374→      </div>
   375→
   376→      {/* Products grid */}
   377→      {isLoading ? (
   378→        <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-3">
   379→          {Array.from({ length: 6 }).map((_, i) => <Skeleton key={i} className="h-72 rounded-xl" />)}
   380→        </div>
   381→      ) : filtered.length === 0 ? (
   382→        <Card><CardContent className="flex flex-col items-center justify-center py-16 text-center">
   383→          <Package className="h-12 w-12 text-muted-foreground/40" />
   384→          <p className="mt-3 text-sm font-medium">No se encontraron productos</p>
   385→          <p className="text-xs text-muted-foreground">Prueba con otra búsqueda o solicita un producto personalizado</p>
   386→        </CardContent></Card>
   387→      ) : (
   388→        <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-3">
   389→          {filtered.map((p) => {
   390→            const savings = p.estimatedCost && p.suggestedPrice ? Math.round(((p.suggestedPrice - p.estimatedCost) / p.suggestedPrice) * 100) : null
   391→            return (
   392→              <div key={p.id} className="group flex cursor-pointer flex-col overflow-hidden rounded-2xl border bg-card shadow-sm transition-all hover:-translate-y-1 hover:shadow-xl" onClick={() => onProductClick(p)}>
   393→                <div className="relative aspect-[4/3] overflow-hidden bg-gradient-to-br from-muted to-muted/50">
   394→                  {p.imageUrl ? (
