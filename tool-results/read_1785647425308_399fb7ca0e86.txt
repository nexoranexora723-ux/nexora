    34→const PAGE_SIZE = 24
    35→
    36→export function CatalogView({ onNavigate, onRegister, onProductClick }: CatalogViewProps) {
    37→  const [query, setQuery] = useState('')
    38→  const [categoryId, setCategoryId] = useState<string | null>(null) // null = all categories
    39→  const [brandFilter, setBrandFilter] = useState<string>('all')
    40→  const [selectedProduct, setSelectedProduct] = useState<Product | null>(null)
    41→  const [imageSearching, setImageSearching] = useState(false)
    42→  const [imageSearchInfo, setImageSearchInfo] = useState<{ description: string; keywords: string[]; count: number } | null>(null)
    43→  const fileInputRef = useRef<HTMLInputElement>(null)
    44→  const { toast } = useToast()
    45→
    46→  // Fetch categories
    47→  const { data: categoriesData } = useQuery<{ id: string; name: string; icon: string | null; productCount: number }[]>({
    48→    queryKey: ['categories-list'],
    49→    queryFn: async () => (await fetch('/api/categories')).json(),
    50→  })
    51→  const categories = categoriesData ?? []
    52→
    53→  // CASCADE: Fetch brands for selected category (or all brands if no category selected)
    54→  const { data: brandsData } = useQuery<{ id: string; name: string; productCount?: number }[] | { brands: { id: string; name: string }[] }>({
    55→    queryKey: ['brands-list', categoryId],
    56→    queryFn: async () => {
    57→      if (categoryId) {
    58→        // Cascade: get brands for this category only
    59→        return await (await fetch(`/api/categories/${categoryId}/brands`)).json()
    60→      }
    61→      // No category selected: get all brands
    62→      return await (await fetch('/api/brands')).json()
    63→    },
    64→    enabled: true,
    65→  })
    66→  // Normalize brands array (cascade returns array, /api/brands returns {brands: [...]})
    67→  const brands = useMemo<{ id: string; name: string; productCount?: number }[]>(() => {
    68→    if (!brandsData) return []
    69→    if (Array.isArray(brandsData)) return brandsData
    70→    return (brandsData.brands ?? []).map((b) => ({ id: b.id, name: b.name }))
    71→  }, [brandsData])
    72→
    73→  // Reset brand filter when category changes
    74→  useEffect(() => {
    75→    setBrandFilter('all')
    76→  }, [categoryId])
    77→
    78→  const {
    79→    data,
    80→    isLoading,
    81→    fetchNextPage,
    82→    hasNextPage,
    83→    isFetchingNextPage,
    84→  } = useInfiniteQuery<{
    85→    products: Product[]
    86→    total: number
    87→    page: number
    88→    totalPages: number
    89→  }>({
    90→    queryKey: ['products-public-infinite', categoryId, brandFilter],
    91→    queryFn: async ({ pageParam = 1 }) => {
    92→      const params = new URLSearchParams()
    93→      params.set('limit', String(PAGE_SIZE))
    94→      params.set('page', String(pageParam))
    95→      if (categoryId) params.set('categoryId', categoryId)
    96→      if (brandFilter !== 'all') params.set('brandId', brandFilter)
    97→      const res = await fetch(`/api/products?${params}`)
    98→      return res.json()
    99→    },
   100→    initialPageParam: 1,
   101→    getNextPageParam: (lastPage) => {
   102→      return lastPage.page < lastPage.totalPages ? lastPage.page + 1 : undefined
   103→    },
   104→  })
   105→
   106→  // Flatten all pages into one array
   107→  const allProducts = useMemo(() => {
   108→    return data?.pages.flatMap((page) => page.products) ?? []
   109→  }, [data])
   110→
   111→  const total = data?.pages[0]?.total ?? 0
   112→
   113→  // Client-side filter by search query only (category and brand are handled by API)
   114→  const filtered = useMemo(() => {
   115→    if (!query) return allProducts
   116→    return allProducts.filter((p) =>
   117→      p.name.toLowerCase().includes(query.toLowerCase()) ||
   118→      p.description?.toLowerCase().includes(query.toLowerCase())
   119→    )
   120→  }, [allProducts, query])
   121→
   122→  // Infinite scroll observer
   123→  useEffect(() => {
   124→    const handleScroll = () => {
   125→      if (
   126→        window.innerHeight + document.documentElement.scrollTop <
   127→        document.documentElement.offsetHeight - 500
   128→      ) return
   129→      if (hasNextPage && !isFetchingNextPage) {
   130→        fetchNextPage()
   131→      }
   132→    }
   133→    window.addEventListener('scroll', handleScroll)
   134→    return () => window.removeEventListener('scroll', handleScroll)
   135→  }, [hasNextPage, isFetchingNextPage, fetchNextPage])
   136→
   137→  return (
   138→    <div className="flex min-h-screen flex-col bg-gradient-to-b from-background to-muted/20">
   139→      <nav className="sticky top-0 z-50 border-b bg-background/80 backdrop-blur-md">
   140→        <div className="mx-auto flex h-16 max-w-7xl items-center justify-between px-4 sm:px-6 lg:px-8">
   141→          <button onClick={() => onNavigate('landing')} className="flex items-center gap-2 text-sm font-medium text-muted-foreground hover:text-foreground">
   142→            <ArrowLeft className="h-4 w-4" /> Volver
   143→          </button>
   144→          <div className="flex items-center gap-2.5">
   145→            <div className="flex h-8 w-8 items-center justify-center rounded-lg overflow-hidden">
   146→              <img src="/icons/logo-official.png" alt="NEXORA" className="h-full w-full object-cover" />
   147→            </div>
   148→            <span className="font-bold">NEXORA</span>
   149→          </div>
   150→          <div className="flex items-center gap-1 sm:gap-2">
   151→            <WishlistCounter />
   152→            <CartCounter />
   153→            {/* Language toggle removed - texts are in Spanish only */}
   154→            <ThemeToggle />
   155→            <Button size="sm" onClick={onRegister}>Registrarse</Button>
   156→          </div>
   157→        </div>
   158→      </nav>
   159→
   160→      <div className="mx-auto w-full max-w-7xl flex-1 px-4 py-8 sm:px-6 lg:px-8">
   161→        {/* Hero del catálogo */}
   162→        <div className="mb-8 rounded-2xl bg-gradient-to-br from-primary/10 via-emerald-500/5 to-transparent p-6">
   163→          <div className="flex items-center gap-2 mb-2">
   164→            <Badge className="gap-1 bg-primary/15 text-primary"><Flame className="h-3 w-3" /> Productos más solicitados</Badge>
   165→          </div>
   166→          <h1 className="text-3xl font-bold tracking-tight">Catálogo de productos importables</h1>
   167→          <p className="mt-2 text-muted-foreground">Productos verificados desde China con precios de fabricante. Tú eliges, nosotros importamos.</p>
   168→        </div>
   169→
   170→        {/* Búsqueda */}
   171→        <div className="mb-4 flex flex-col gap-2 sm:flex-row sm:items-center">
   172→          <div className="relative max-w-md flex-1">
   173→            <Search className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   174→            <Input placeholder="Buscar productos..." value={query} onChange={(e) => setQuery(e.target.value)} className="pl-9" />
   175→          </div>
   176→          <input
   177→            ref={fileInputRef}
   178→            type="file"
   179→            accept="image/*"
   180→            className="hidden"
   181→            onChange={async (e) => {
   182→              const file = e.target.files?.[0]
   183→              if (!file) return
   184→              setImageSearching(true)
   185→              setImageSearchInfo(null)
   186→              try {
   187→                // Convert to data URL
   188→                const dataUrl = await new Promise<string>((resolve, reject) => {
   189→                  const reader = new FileReader()
   190→                  reader.onload = () => resolve(reader.result as string)
   191→                  reader.onerror = reject
   192→                  reader.readAsDataURL(file)
   193→                })
   194→                const res = await fetch('/api/search-by-image', {
   195→                  method: 'POST',
   196→                  headers: { 'Content-Type': 'application/json' },
   197→                  body: JSON.stringify({ image: dataUrl }),
   198→                })
   199→                const data = await res.json()
   200→                if (!res.ok) throw new Error(data.error ?? 'Error en la búsqueda')
   201→                setImageSearchInfo({
   202→                  description: data.description ?? '',
   203→                  keywords: data.keywords ?? [],
   204→                  count: data.count ?? 0,
   205→                })
   206→                // Apply the first keyword as the live search query so results filter immediately
   207→                if (Array.isArray(data.keywords) && data.keywords.length > 0) {
   208→                  setQuery(data.keywords[0])
   209→                }
   210→                toast({
   211→                  title: 'Búsqueda por imagen completada',
   212→                  description: data.description ?? `${data.count ?? 0} productos encontrados`,
   213→                })
   214→              } catch (err) {
   215→                toast({
   216→                  title: 'Error en búsqueda por imagen',
   217→                  description: err instanceof Error ? err.message : 'Intenta con otra imagen',
   218→                  variant: 'destructive',
   219→                })
   220→              } finally {
   221→                setImageSearching(false)
   222→                if (fileInputRef.current) fileInputRef.current.value = ''
   223→              }
   224→            }}
   225→          />
   226→          <Button
   227→            variant="outline"
   228→            size="sm"
   229→            className="gap-1.5"
   230→            disabled={imageSearching}
   231→            onClick={() => fileInputRef.current?.click()}
   232→          >
   233→            {imageSearching ? (
   234→              <>
   235→                <Loader2 className="h-3.5 w-3.5 animate-spin" /> Analizando...
   236→              </>
   237→            ) : (
   238→              <>
   239→                <ImageIcon className="h-3.5 w-3.5" /> Buscar por imagen
   240→              </>
   241→            )}
   242→          </Button>
   243→        </div>
   244→
   245→        {/* Result info banner after image search */}
   246→        {imageSearchInfo && (
   247→          <div className="mb-4 flex flex-wrap items-center gap-2 rounded-xl border border-primary/20 bg-primary/5 p-3 text-sm">
   248→            <Sparkles className="h-4 w-4 text-primary" />
   249→            <span className="font-medium">{imageSearchInfo.description || 'Búsqueda por imagen'}</span>
   250→            <span className="text-muted-foreground">·</span>
   251→            <span className="text-muted-foreground">{imageSearchInfo.count} coincidencias</span>
   252→            {imageSearchInfo.keywords.length > 0 && (
   253→              <div className="flex flex-wrap items-center gap-1">
   254→                {imageSearchInfo.keywords.slice(0, 6).map((kw) => (
   255→                  <Badge key={kw} variant="secondary" className="text-[10px]">{kw}</Badge>
   256→                ))}
   257→              </div>
   258→            )}
   259→            <Button
   260→              variant="ghost"
   261→              size="sm"
   262→              className="ml-auto h-7 gap-1 text-xs text-muted-foreground"
   263→              onClick={() => { setImageSearchInfo(null); setQuery('') }}
   264→            >
   265→              <X className="h-3 w-3" /> Limpiar
   266→            </Button>
   267→          </div>
   268→        )}
   269→
   270→        {/* Filtros de categoría (cascade: al seleccionar, actualiza las marcas) */}
   271→        <div className="mb-4 flex flex-wrap gap-1.5">
   272→          <Button 
   273→            variant={!categoryId ? 'default' : 'outline'} 
   274→            size="sm" 
   275→            onClick={() => setCategoryId(null)}
   276→          >📦 Todos</Button>
   277→          {categories.map((c) => (
   278→            <Button 
   279→              key={c.id} 
   280→              variant={categoryId === c.id ? 'default' : 'outline'} 
   281→              size="sm" 
   282→              onClick={() => setCategoryId(c.id)}
   283→            >{c.icon} {c.name}</Button>
   284→          ))}
   285→        </div>
   286→
   287→        {/* Filtro de marca (debajo de categorías, con mejor diseño) */}
   288→        <div className="mb-4 flex items-center gap-3 rounded-xl border bg-muted/30 p-3">
   289→          <div className="flex items-center gap-2 text-sm font-medium text-muted-foreground">
   290→            <Filter className="h-4 w-4 text-primary" />
   291→            <span>Filtrar por marca:</span>
   292→          </div>
   293→          <Select value={brandFilter} onValueChange={setBrandFilter}>
   294→            <SelectTrigger className="h-9 w-[220px] border-primary/20 bg-background">
   295→              <SelectValue placeholder={categoryId ? "Marcas en esta categoría" : "Todas las marcas"} />
   296→            </SelectTrigger>
   297→            <SelectContent>
   298→              <SelectItem value="all">{categoryId ? "Todas las marcas (categoría)" : "Todas las marcas"}</SelectItem>
   299→              {brands.map((b) => (
   300→                <SelectItem key={b.id} value={b.id}>
   301→                  {b.name}{b.productCount ? ` (${b.productCount})` : ''}
   302→                </SelectItem>
   303→              ))}
   304→            </SelectContent>
   305→          </Select>
   306→          {brandFilter !== 'all' && (
   307→            <Button 
   308→              variant="ghost" 
   309→              size="sm" 
   310→              className="h-9 gap-1 text-xs text-muted-foreground hover:text-foreground"
   311→              onClick={() => setBrandFilter('all')}
   312→            >
   313→              ✕ Quitar filtro
   314→            </Button>
   315→          )}
   316→          {categoryId && (
   317→            <span className="ml-auto text-xs text-muted-foreground">
   318→              {brands.length} marca{brands.length !== 1 ? 's' : ''} disponible{brands.length !== 1 ? 's' : ''}
   319→            </span>
   320→          )}
   321→        </div>
   322→
   323→        {/* Contador de resultados */}
   324→        <div className="mb-4 flex items-center gap-2 text-sm text-muted-foreground">
   325→          <Badge variant="outline">{total.toLocaleString()} productos disponibles</Badge>
   326→          {brandFilter !== 'all' && (
   327→            <Badge variant="secondary" className="gap-1">
   328→              {brands.find((b) => b.id === brandFilter)?.name}
   329→              <button onClick={() => setBrandFilter('all')} className="ml-1 hover:text-foreground">✕</button>
   330→            </Badge>
   331→          )}
   332→          {filtered.length < total && <span className="text-xs">Mostrando {filtered.length}...</span>}
   333→        </div>
   334→
   335→        {/* Grid de productos */}
   336→        {isLoading ? (
   337→          <div className="grid grid-cols-1 gap-6 sm:grid-cols-2 lg:grid-cols-4">
   338→            {Array.from({ length: 8 }).map((_, i) => <Skeleton key={i} className="h-80 rounded-xl" />)}
   339→          </div>
   340→        ) : filtered.length === 0 ? (
   341→          <Card><CardContent className="flex flex-col items-center justify-center py-16 text-center">
   342→            <Package className="h-12 w-12 text-muted-foreground/40" />
   343→            <p className="mt-3 text-sm font-medium">No se encontraron productos</p>
   344→            <Button className="mt-4" onClick={onRegister}>Solicitar producto personalizado</Button>
   345→          </CardContent></Card>
   346→        ) : (
   347→          <>
   348→            <div className="grid grid-cols-1 gap-6 sm:grid-cols-2 lg:grid-cols-3 xl:grid-cols-4">
   349→              {filtered.map((p) => (
   350→                <ProductCard key={p.id} product={p} onClick={() => onProductClick ? onProductClick(p.id) : setSelectedProduct(p)} />
   351→              ))}
   352→            </div>
   353→
   354→            {/* Loading more indicator */}
   355→            {isFetchingNextPage && (
   356→              <div className="mt-8 flex justify-center">
   357→                <Loader2 className="h-6 w-6 animate-spin text-primary" />
   358→                <span className="ml-2 text-sm text-muted-foreground">Cargando más productos...</span>
   359→              </div>
   360→            )}
   361→
   362→            {/* Load more button (fallback for scroll) */}
   363→            {hasNextPage && !isFetchingNextPage && (
   364→              <div className="mt-8 flex justify-center">
   365→                <Button variant="outline" onClick={() => fetchNextPage()}>
   366→                  Cargar más productos
   367→                </Button>
   368→              </div>
   369→            )}
   370→
   371→            {/* End of list */}
   372→            {!hasNextPage && filtered.length > 0 && (
   373→              <div className="mt-8 text-center text-sm text-muted-foreground">
   374→                ✅ Has visto todos los {total.toLocaleString()} productos
   375→              </div>
   376→            )}
   377→          </>
   378→        )}
   379→      </div>
   380→
   381→      {/* Product detail dialog */}
   382→      {selectedProduct && (
   383→        <ProductDetailDialog product={selectedProduct} onClose={() => setSelectedProduct(null)} onRequest={onRegister} />
   384→      )}
   385→
   386→      {/* Footer */}
   387→      <SiteFooter onNavigate={onNavigate} />
   388→
   389→      {/* Drawers */}
   390→      <CartDrawer />
   391→      <WishlistDrawer />
   392→
   393→      {/* Product comparator */}
   394→      <CompareProducts />
   395→    </div>
   396→  )
   397→}

   399→// === Product Card — con favoritos, compartir, carrito ===
   400→function ProductCard({ product, onClick }: { product: Product; onClick: () => void }) {
   401→  const savings = product.estimatedCost && product.suggestedPrice
   402→    ? Math.round(((product.suggestedPrice - product.estimatedCost) / product.suggestedPrice) * 100)
   403→    : null
   404→  
   405→  const wishlist = useWishlist()
   406→  const cart = useCart()
   407→  const { toast } = useToast()
   408→  const isInWishlist = wishlist.has(product.id)
   409→
   410→  const handleWishlist = (e: React.MouseEvent) => {
   411→    e.stopPropagation()
   412→    wishlist.toggle({
   413→      id: product.id,
   414→      name: product.name,
   415→      imageUrl: product.imageUrl,
   416→      price: product.estimatedCost ?? 0,
   417→      sku: product.sku,
   418→    })
   419→    toast({ title: isInWishlist ? 'Removido de favoritos' : 'Añadido a favoritos' })
   420→  }
   421→
   422→  const handleShare = (e: React.MouseEvent) => {
   423→    e.stopPropagation()
   424→    const url = `${window.location.origin}/producto/${product.id}`
   425→    const text = `Mira este producto en NEXORA: ${product.name} - $${product.estimatedCost}`
   426→    if (navigator.share) {
   427→      navigator.share({ title: product.name, text, url })
   428→    } else {
   429→      navigator.clipboard.writeText(`${text} ${url}`)
   430→      toast({ title: 'Link copiado al portapapeles' })
   431→    }
   432→  }
   433→
   434→  const handleAddToCart = (e: React.MouseEvent) => {
   435→    e.stopPropagation()
   436→    cart.addItem({
   437→      id: product.id,
   438→      name: product.name,
   439→      imageUrl: product.imageUrl,
   440→      price: product.estimatedCost ?? 0,
   441→      sku: product.sku,
   442→    })
   443→    toast({ title: 'Añadido al carrito' })
   444→  }
   445→
   446→  return (
   447→    <div
   448→      className="group flex cursor-pointer flex-col overflow-hidden rounded-2xl border bg-card shadow-sm transition-all hover:-translate-y-1 hover:shadow-xl"
   449→      onClick={onClick}
   450→    >
   451→      {/* Imagen con badges */}
   452→      <div className="relative aspect-[4/3] overflow-hidden bg-gradient-to-br from-muted to-muted/50">
   453→        {product.imageUrl ? (
   454→          <img
   455→            src={product.imageUrl}
   456→            alt={product.name}
   457→            className="h-full w-full object-cover transition-transform duration-500 group-hover:scale-110"
   458→            loading="lazy"
   459→            decoding="async"
   460→            onError={(e) => { (e.currentTarget as HTMLImageElement).src = '/products/placeholder.svg' }}
   461→          />
   462→        ) : (
   463→          <img
   464→            src="/products/placeholder.svg"
   465→            alt={product.name}
   466→            className="h-full w-full object-cover"
   467→            loading="lazy"
   468→            decoding="async"
   469→          />
   470→        )}
   471→        <div className="absolute inset-0 bg-gradient-to-t from-black/20 to-transparent opacity-0 transition-opacity group-hover:opacity-100" />
   472→        
   473→        {/* Botones flotantes (aparecen en hover) */}
   474→        <div className="absolute right-2 top-2 flex flex-col gap-1.5 opacity-0 transition-opacity group-hover:opacity-100">
   475→          <button
   476→            onClick={handleWishlist}
   477→            className={cn(
   478→              'flex h-8 w-8 items-center justify-center rounded-full shadow-lg backdrop-blur transition-colors',
   479→              isInWishlist ? 'bg-rose-500 text-white' : 'bg-background/90 text-foreground hover:bg-background'
   480→            )}
   481→          >
   482→            <Heart className={cn('h-4 w-4', isInWishlist && 'fill-white')} />
   483→          </button>
   484→          <button
   485→            onClick={handleShare}
   486→            className="flex h-8 w-8 items-center justify-center rounded-full bg-background/90 text-foreground shadow-lg backdrop-blur transition-colors hover:bg-background"
   487→          >
   488→            <Share2 className="h-4 w-4" />
   489→          </button>
   490→          <CompareToggleButton item={toCompareItem(product)} />
   491→        </div>
   492→
   493→        {/* Badge de categoría */}
   494→        {product.category?.icon && (
   495→          <div className="absolute left-3 top-3 flex items-center gap-1.5 rounded-full bg-background/90 px-2.5 py-1 text-xs font-medium shadow-sm backdrop-blur">
   496→            <span>{product.category.icon}</span>
   497→            <span className="text-foreground">{product.category.name}</span>
