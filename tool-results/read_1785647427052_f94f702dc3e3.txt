   399→   399→// === Product Card — con favoritos, compartir, carrito ===
   400→   400→function ProductCard({ product, onClick }: { product: Product; onClick: () => void }) {
   401→   401→  const savings = product.estimatedCost && product.suggestedPrice
   402→   402→    ? Math.round(((product.suggestedPrice - product.estimatedCost) / product.suggestedPrice) * 100)
   403→   403→    : null
   404→   404→  
   405→   405→  const wishlist = useWishlist()
   406→   406→  const cart = useCart()
   407→   407→  const { toast } = useToast()
   408→   408→  const isInWishlist = wishlist.has(product.id)
   409→   409→
   410→   410→  const handleWishlist = (e: React.MouseEvent) => {
   411→   411→    e.stopPropagation()
   412→   412→    wishlist.toggle({
   413→   413→      id: product.id,
   414→   414→      name: product.name,
   415→   415→      imageUrl: product.imageUrl,
   416→   416→      price: product.estimatedCost ?? 0,
   417→   417→      sku: product.sku,
   418→   418→    })
   419→   419→    toast({ title: isInWishlist ? 'Removido de favoritos' : 'Añadido a favoritos' })
