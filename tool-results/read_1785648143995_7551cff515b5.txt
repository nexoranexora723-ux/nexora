   240→function AdminRequests({ selectedId, onSelect }: { selectedId: string | null; onSelect: (id: string) => void }) {
   241→  const [query, setQuery] = useState('')
   242→  const [statusFilter, setStatusFilter] = useState('all')
   243→  const { data: requests, isLoading } = useQuery<ImportRequest[]>({
   244→    queryKey: ['admin-requests', query, statusFilter],
   245→    queryFn: async () => {
   246→      const params = new URLSearchParams()
   247→      if (query) params.set('q', query)
   248→      if (statusFilter !== 'all') params.set('status', statusFilter)
   249→      return (await fetch(`/api/requests?${params}`)).json()
   250→    },
   251→  })
   252→
   253→  if (selectedId) return <RequestDetail id={selectedId} onBack={() => onSelect('')} />
   254→
   255→  return (
   256→    <div className="space-y-4">
   257→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center">
   258→        <div className="relative max-w-sm flex-1">
   259→          <Search className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   260→          <Input placeholder="Buscar por número o producto..." value={query} onChange={(e) => setQuery(e.target.value)} className="pl-9" />
   261→        </div>
   262→        <div className="flex flex-wrap gap-1.5">
   263→          {['all', 'NUEVA', 'BUSCANDO_PROVEEDOR', 'COTIZACION_ENVIADA', 'PAGO_RECIBIDO', 'EN_TRANSITO', 'ENTREGADO'].map((s) => (
   264→            <Button key={s} variant={statusFilter === s ? 'default' : 'outline'} size="sm" className="h-8 text-xs" onClick={() => setStatusFilter(s)}>
   265→              {s === 'all' ? 'Todas' : REQUEST_STATUS_LABELS[s]}
   266→            </Button>

   268→        </div>

   271→      {isLoading ? (
   272→        <div className="space-y-2">{Array.from({ length: 6 }).map((_, i) => <Skeleton key={i} className="h-16 w-full" />)}</div>

   279→              <TableHeader><TableRow>
   280→                <TableHead>Número</TableHead><TableHead>Producto</TableHead><TableHead>Cliente</TableHead>

   282→              </TableRow></TableHeader>

   284→                {requests!.map((r) => (
   285→                  <TableRow key={r.id} className="group cursor-pointer" onClick={() => onSelect(r.id)}>
   286→                    <TableCell><code className="text-xs font-medium text-primary">{r.number}</code></TableCell>
   287→                    <TableCell><p className="truncate text-sm font-medium">{r.productName}</p></TableCell>
   288→                    <TableCell className="text-sm text-muted-foreground">{r.client.firstName} {r.client.lastName}</TableCell>
   289→                    <TableCell className="text-right text-sm tabular-nums">{r.quantity}</TableCell>
   290→                    <TableCell><Badge variant="outline" className={cn('text-[10px]', REQUEST_STATUS_COLORS[r.status])}>{REQUEST_STATUS_LABELS[r.status]}</Badge></TableCell>

   304→// === Request Detail ===
   305→function RequestDetail({ id, onBack }: { id: string; onBack: () => void }) {
   306→  const qc = useQueryClient()
   307→  const { data: req, isLoading } = useQuery<ImportRequest>({
   308→    queryKey: ['request', id],
   309→    queryFn: async () => (await fetch(`/api/requests/${id}`)).json(),
   310→  })
   311→  const [statusOpen, setStatusOpen] = useState(false)
   312→  const [newStatus, setNewStatus] = useState('')
   313→  const [notes, setNotes] = useState('')
   314→
   315→  const updateStatus = useMutation({
   316→    mutationFn: async () => {
   317→      await fetch(`/api/requests/${id}/status`, {
   318→        method: 'PATCH', headers: { 'Content-Type': 'application/json' },
   319→        body: JSON.stringify({ status: newStatus, notes }),
   320→      })
   321→    },
   322→    onSuccess: () => {
   323→      qc.invalidateQueries({ queryKey: ['request', id] })
   324→      qc.invalidateQueries({ queryKey: ['admin-requests'] })
   325→      setStatusOpen(false)
   326→      setNotes('')
   327→    },
   328→  })
   329→
   330→  if (isLoading) return <Skeleton className="h-96 w-full" />
   331→  if (!req) return <Card><CardContent className="py-12 text-center">No encontrada</CardContent></Card>
   332→
   333→  return (
   334→    <div className="space-y-6">
   335→      <button onClick={onBack} className="flex items-center gap-1 text-sm text-muted-foreground hover:text-foreground">← Volver</button>
   336→      <div className="flex items-start justify-between">
   337→        <div>
   338→          <div className="flex items-center gap-2">
   339→            <code className="text-sm font-medium text-primary">{req.number}</code>
   340→            <Badge variant="outline" className={REQUEST_STATUS_COLORS[req.status]}>{REQUEST_STATUS_LABELS[req.status]}</Badge>
   341→          </div>
   342→          <h2 className="mt-1 text-2xl font-bold">{req.productName}</h2>
   343→          <p className="text-sm text-muted-foreground">Cliente: {req.client.firstName} {req.client.lastName} ({req.client.email})</p>
   344→        </div>
   345→        <Button onClick={() => { setNewStatus(req.status); setStatusOpen(true) }}>Cambiar estado</Button>
   346→      </div>
   347→
   348→      <div className="grid grid-cols-1 gap-4 lg:grid-cols-3">
   349→        <Card className="lg:col-span-2"><CardContent className="p-5">
   350→          <h3 className="mb-3 text-sm font-semibold">Información de la solicitud</h3>
   351→          <div className="grid grid-cols-2 gap-4 text-sm">
   352→            <div><p className="text-xs text-muted-foreground">Cantidad</p><p className="font-medium">{req.quantity} unidades</p></div>
   353→            <div><p className="text-xs text-muted-foreground">Presupuesto</p><p className="font-medium">{req.budget ? formatCurrency(req.budget) : '—'}</p></div>
   354→            <div><p className="text-xs text-muted-foreground">Categoría</p><p className="font-medium">{req.category ?? '—'}</p></div>
   355→            <div><p className="text-xs text-muted-foreground">Propósito</p><p className="font-medium">{req.purpose ?? '—'}</p></div>
   356→            <div><p className="text-xs text-muted-foreground">Prioridad</p><p className="font-medium">{req.priority}</p></div>
