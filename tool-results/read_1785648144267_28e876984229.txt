   273→// === Product Form Dialog with tabs ===
   274→function ProductFormDialog({ open, onOpenChange, product }: { open: boolean; onOpenChange: (o: boolean) => void; product: AdminProduct | null }) {
   275→  const qc = useQueryClient()
   276→  const { toast } = useToast()
   277→  const isEdit = !!product
   278→
   279→  const { data: catalog } = useQuery<CatalogData>({
   280→    queryKey: ['admin-catalog'],
   281→    queryFn: async () => (await fetch('/api/admin/catalog')).json(),
   282→  })
   283→
   284→  // Form state
   285→  const [form, setForm] = useState({
   286→    sku: '', name: '', description: '', longDescription: '',
   287→    brandId: '', categoryId: '', supplierId: '',
   288→    imageUrl: '', images: [] as string[], videoUrl: '',
   289→    estimatedCost: '', suggestedPrice: '', currencyCode: 'USD',
   290→    status: 'ACTIVE', isFeatured: false,
   291→    specs: [] as { label: string; value: string }[],
   292→    features: [] as string[],
   293→  })
   294→  const [error, setError] = useState('')
   295→  const [loading, setLoading] = useState(false)
   296→  const [showSuccess, setShowSuccess] = useState(false)
   297→  const [activeTab, setActiveTab] = useState('basic')
   298→  const [newFeature, setNewFeature] = useState('')
   299→  const [newSpecLabel, setNewSpecLabel] = useState('')
   300→  const [newSpecValue, setNewSpecValue] = useState('')
   301→
   302→  // Sync form when dialog opens
   303→  useEffect(() => {
   304→    if (open) {
   305→      setError('')
   306→      if (product) {
   307→        setForm({
   308→          sku: product.sku, name: product.name,
   309→          description: product.description ?? '', longDescription: product.longDescription ?? '',
   310→          brandId: product.brand?.id ?? '', categoryId: product.category?.id ?? '', supplierId: product.supplier?.id ?? '',
   311→          imageUrl: product.imageUrl ?? '', images: product.images, videoUrl: product.videoUrl ?? '',
   312→          estimatedCost: product.estimatedCost?.toString() ?? '', suggestedPrice: product.suggestedPrice?.toString() ?? '',
   313→          currencyCode: product.currencyCode, status: product.status, isFeatured: product.isFeatured,
   314→          specs: product.specs, features: product.features,
   315→        })
   316→      } else {
   317→        setForm({
   318→          sku: '', name: '', description: '', longDescription: '',
   319→          brandId: '', categoryId: '', supplierId: '',
   320→          imageUrl: '', images: [], videoUrl: '',
   321→          estimatedCost: '', suggestedPrice: '', currencyCode: 'USD',
   322→          status: 'ACTIVE', isFeatured: false, specs: [], features: [],
   323→        })
   324→      }
   325→      setActiveTab('basic')
   326→    }
   327→  }, [open, product])
   328→
   329→  const save = useMutation({
   330→    mutationFn: async () => {
   331→      const payload = {
   332→        ...form,
   333→        estimatedCost: form.estimatedCost ? Number(form.estimatedCost) : null,
   334→        suggestedPrice: form.suggestedPrice ? Number(form.suggestedPrice) : null,
   335→        brandId: form.brandId || null,
   336→        categoryId: form.categoryId || null,
   337→        supplierId: form.supplierId || null,
   338→      }
   339→      const url = isEdit ? `/api/admin/products/${product!.id}` : '/api/admin/products'
   340→      const method = isEdit ? 'PUT' : 'POST'
   341→      const res = await fetch(url, { method, headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload) })
   342→      if (!res.ok) {
   343→        const data = await res.json()
   344→        throw new Error(data.error || 'Error')
   345→      }
   346→    },
   347→    onSuccess: () => {
   348→      qc.invalidateQueries({ queryKey: ['admin-products'] })
   349→      qc.invalidateQueries({ queryKey: ['products-public'] })
   350→      qc.invalidateQueries({ queryKey: ['products'] })
   351→      if (!isEdit) fireConfetti()
   352→      setShowSuccess(true)
   353→    },
   354→  })
   355→
   356→  const handleSubmit = async () => {
   357→    setError('')
   358→    if (!form.sku.trim() || !form.name.trim()) { setError('SKU y nombre son obligatorios'); setActiveTab('basic'); return }
   359→    setLoading(true)
   360→    try {
   361→      await save.mutateAsync()
   362→    } catch (err) {
   363→      setError(err instanceof Error ? err.message : 'Error')
   364→    } finally {
   365→      setLoading(false)
   366→    }
   367→  }
   368→
   369→  // Upload handler
   370→  const fileRef = useRef<HTMLInputElement>(null)
   371→  const [uploading, setUploading] = useState(false)
   372→
   373→  const handleUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
   374→    const file = e.target.files?.[0]
   375→    if (!file) return
   376→    setUploading(true)
   377→    try {
   378→      const formData = new FormData()
   379→      formData.append('file', file)
   380→      const res = await fetch('/api/admin/upload', { method: 'POST', body: formData })
   381→      const data = await res.json()
   382→      if (!res.ok) throw new Error(data.error)
   383→      setForm((f) => ({ ...f, imageUrl: data.url, images: [...f.images, data.url] }))
   384→      toast({ title: 'Imagen subida', description: data.url })
   385→    } catch (err) {
   386→      toast({ title: 'Error al subir', description: err instanceof Error ? err.message : 'Error', variant: 'destructive' })
   387→    } finally {
   388→      setUploading(false)
   389→    }
   390→  }
   391→
   392→  const addImageUrl = (url: string) => {
   393→    if (url.trim()) {
   394→      setForm((f) => ({ ...f, images: [...f.images, url.trim()], imageUrl: f.imageUrl || url.trim() }))
   395→    }
   396→  }
   397→
   398→  const removeImage = (idx: number) => {
   399→    setForm((f) => {
   400→      const newImages = f.images.filter((_, i) => i !== idx)
   401→      return { ...f, images: newImages, imageUrl: newImages[0] ?? '' }
   402→    })
   403→  }
   404→
   405→  const moveImage = (idx: number, dir: 'up' | 'down') => {
   406→    setForm((f) => {
   407→      const newImages = [...f.images]
   408→      const target = dir === 'up' ? idx - 1 : idx + 1
   409→      if (target < 0 || target >= newImages.length) return f
   410→      ;[newImages[idx], newImages[target]] = [newImages[target], newImages[idx]]
   411→      return { ...f, images: newImages, imageUrl: newImages[0] ?? '' }
   412→    })
   413→  }
   414→
   415→  const addFeature = () => {
   416→    if (newFeature.trim()) {
   417→      setForm((f) => ({ ...f, features: [...f.features, newFeature.trim()] }))
   418→      setNewFeature('')
   419→    }
   420→  }
   421→
   422→  const addSpec = () => {
   423→    if (newSpecLabel.trim() && newSpecValue.trim()) {
   424→      setForm((f) => ({ ...f, specs: [...f.specs, { label: newSpecLabel.trim(), value: newSpecValue.trim() }] }))
   425→      setNewSpecLabel(''); setNewSpecValue('')
   426→    }
   427→  }
   428→
   429→  return (
   430→    <>
   431→      <Dialog open={open && !showSuccess} onOpenChange={onOpenChange}>
   432→        <DialogContent className="nexora-scroll max-h-[92vh] overflow-y-auto sm:max-w-2xl">
   433→          <DialogHeader>
   434→            <DialogTitle className="flex items-center gap-2"><Package className="h-5 w-5 text-primary" /> {isEdit ? 'Editar producto' : 'Nuevo producto'}</DialogTitle>
   435→            <DialogDescription>{isEdit ? `Modificando ${product?.sku}` : 'Completa la información del producto'}</DialogDescription>
   436→          </DialogHeader>
   437→
   438→          <Tabs value={activeTab} onValueChange={setActiveTab}>
   439→            <TabsList className="grid w-full grid-cols-3 sm:grid-cols-6">
   440→              <TabsTrigger value="basic" className="text-xs">Info</TabsTrigger>
