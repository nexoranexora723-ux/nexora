   297→   297→// === Product Card — con favoritos, compartir, carrito ===
   298→   298→function ProductCard({ product, onClick }: { product: Product; onClick: () => void }) {
   299→   299→  const savings = product.estimatedCost && product.suggestedPrice
   300→   300→    ? Math.round(((product.suggestedPrice - product.estimatedCost) / product.suggestedPrice) * 100)
   301→   301→    : null
   302→   302→  
   303→   303→  const wishlist = useWishlist()
   304→   304→  const cart = useCart()
   305→   305→  const { toast } = useToast()
   306→   306→  const isInWishlist = wishlist.has(product.id)
   307→   307→
   308→   308→  const handleWishlist = (e: React.MouseEvent) => {
   309→   309→    e.stopPropagation()
   310→   310→    wishlist.toggle({
   311→   311→      id: product.id,
   312→   312→      name: product.name,
   313→   313→      imageUrl: product.imageUrl,
   314→   314→      price: product.estimatedCost ?? 0,
   315→   315→      sku: product.sku,
   316→   316→    })
   317→   317→    toast({ title: isInWishlist ? 'Removido de favoritos' : 'Añadido a favoritos' })
   318→   318→  }
   319→   319→
   320→   320→  const handleShare = (e: React.MouseEvent) => {
   321→   321→    e.stopPropagation()
   322→   322→    const url = `${window.location.origin}/producto/${product.id}`
   323→   323→    const text = `Mira este producto en NEXORA: ${product.name} - $${product.estimatedCost}`
   324→   324→    if (navigator.share) {
   325→   325→      navigator.share({ title: product.name, text, url })
   326→   326→    } else {
   327→   327→      navigator.clipboard.writeText(`${text} ${url}`)
   328→   328→      toast({ title: 'Link copiado al portapapeles' })
   329→   329→    }
   330→   330→  }
   331→   331→
   332→   332→  const handleAddToCart = (e: React.MouseEvent) => {
   333→   333→    e.stopPropagation()
   334→   334→    cart.addItem({
   335→   335→      id: product.id,
   336→   336→      name: product.name,
   337→   337→      imageUrl: product.imageUrl,
   338→   338→      price: product.estimatedCost ?? 0,
   339→   339→      sku: product.sku,
   340→   340→    })
   341→   341→    toast({ title: 'Añadido al carrito' })
   342→   342→  }
   343→   343→
   344→   344→  return (
   345→   345→    <div
   346→   346→      className="group flex cursor-pointer flex-col overflow-hidden rounded-2xl border bg-card shadow-sm transition-all hover:-translate-y-1 hover:shadow-xl"
   347→   347→      onClick={onClick}
   348→   348→    >
   349→   349→      {/* Imagen con badges */}
   350→   350→      <div className="relative aspect-[4/3] overflow-hidden bg-gradient-to-br from-muted to-muted/50">
   351→   351→        {product.imageUrl ? (
   352→   352→          <img
   353→   353→            src={product.imageUrl}
   354→   354→            alt={product.name}
   355→   355→            className="h-full w-full object-cover transition-transform duration-500 group-hover:scale-110"
   356→   356→            loading="lazy"
   357→   357→          />
   358→   358→        ) : (
   359→   359→          <div className="flex h-full w-full items-center justify-center text-5xl">📦</div>
   360→   360→        )}
   361→   361→        <div className="absolute inset-0 bg-gradient-to-t from-black/20 to-transparent opacity-0 transition-opacity group-hover:opacity-100" />
   362→   362→        
   363→   363→        {/* Botones flotantes (aparecen en hover) */}
   364→   364→        <div className="absolute right-2 top-2 flex flex-col gap-1.5 opacity-0 transition-opacity group-hover:opacity-100">
   365→   365→          <button
   366→   366→            onClick={handleWishlist}
   367→   367→            className={cn(
   368→   368→              'flex h-8 w-8 items-center justify-center rounded-full shadow-lg backdrop-blur transition-colors',
   369→   369→              isInWishlist ? 'bg-rose-500 text-white' : 'bg-background/90 text-foreground hover:bg-background'
   370→   370→            )}
   371→   371→          >
   372→   372→            <Heart className={cn('h-4 w-4', isInWishlist && 'fill-white')} />
   373→   373→          </button>
   374→   374→          <button
   375→   375→            onClick={handleShare}
   376→   376→            className="flex h-8 w-8 items-center justify-center rounded-full bg-background/90 text-foreground shadow-lg backdrop-blur transition-colors hover:bg-background"
   377→   377→          >
   378→   378→            <Share2 className="h-4 w-4" />
   379→   379→          </button>
   380→   380→        </div>
   381→   381→
   382→   382→        {/* Badge de categoría */}
   383→   383→        {product.category?.icon && (
   384→   384→          <div className="absolute left-3 top-3 flex items-center gap-1.5 rounded-full bg-background/90 px-2.5 py-1 text-xs font-medium shadow-sm backdrop-blur">
   385→   385→            <span>{product.category.icon}</span>
   386→   386→            <span className="text-foreground">{product.category.name}</span>
   387→   387→          </div>
   388→   388→        )}
   389→   389→        {savings && savings > 0 && (
   390→   390→          <div className="absolute right-3 top-3 flex items-center gap-1 rounded-full bg-rose-500 px-2.5 py-1 text-xs font-bold text-white shadow-lg group-hover:opacity-0">
   391→   391→            <TrendingUp className="h-3 w-3" /> {savings}% OFF
   392→   392→          </div>
   393→   393→        )}
   394→   394→        {product.isFeatured && (
   395→   395→          <div className="absolute bottom-3 left-3 flex items-center gap-1 rounded-full bg-amber-500 px-2.5 py-1 text-xs font-bold text-white shadow-lg">
   396→   396→            <Star className="h-3 w-3 fill-white" /> Destacado
   397→   397→          </div>
   398→   398→        )}
   399→   399→      </div>
   400→   400→
   401→   401→      {/* Contenido */}
   402→   402→      <div className="flex flex-1 flex-col p-4">
   403→   403→        {product.brand && <p className="text-xs font-medium text-muted-foreground">{product.brand.name}</p>}
   404→   404→        <h3 className="mt-0.5 line-clamp-2 text-sm font-semibold leading-snug">{product.name}</h3>
   405→   405→        {product.description && <p className="mt-1 line-clamp-2 text-xs text-muted-foreground">{product.description}</p>}
